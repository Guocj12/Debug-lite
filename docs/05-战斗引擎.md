# 战斗引擎设计

## 一、概述

战斗引擎运行在服务端，是游戏的唯一真理来源。客户端只负责发送指令和渲染结果。

## 二、核心模型

### Player（玩家状态）

```javascript
class Player {
    constructor(id, name, characterConfig) {
        this.id = id;                    // 'p1' | 'p2'
        this.name = name;                // 玩家昵称
        this.characterId = characterConfig.id;
        
        // 战斗属性
        this.maxHp = characterConfig.baseHp;
        this.hp = this.maxHp;
        this.baseAttack = characterConfig.baseAttack;
        this.baseDefense = characterConfig.baseDefense;
        this.speed = characterConfig.speed;
        
        // 战场状态
        this.position = { x: 0, y: 5 };  // 默认位置
        this.isAlive = true;
        this.isDefending = false;
        
        // 技能
        this.selectedSkills = [];         // 本局选择的技能ID列表
        this.skillCooldowns = {};         // { skillId: remainingCooldown }
        
        // Buff/Debuff
        this.buffs = [];                  // [{ type, value, remainingDuration }]
        
        // 行动队列
        this.actionQueue = [];            // 本回合的行动队列
        this.queueConfirmed = false;
        
        // 统计
        this.stats = {
            damageDealt: 0,
            damageTaken: 0,
            healingDone: 0,
            skillsUsed: 0
        };
    }
}
```

## 三、战斗流程

### 回合执行伪代码

```javascript
function executeRound(playerA, playerB, roomId) {
    const maxLen = Math.max(playerA.actionQueue.length, playerB.actionQueue.length);
    const battleLog = [];
    
    for (let step = 0; step < maxLen; step++) {
        const actionsThisStep = [];
        const resultsThisStep = [];
        
        // === 第1步：收集双方当前步骤的行动 ===
        const aAction = playerA.actionQueue[step] || { action: 'idle' };
        const bAction = playerB.actionQueue[step] || { action: 'idle' };
        
        // 眩晕检查
        const aStunned = hasBuff(playerA, 'stun');
        const bStunned = hasBuff(playerB, 'stun');
        
        const effectiveA = aStunned ? { action: 'idle' } : aAction;
        const effectiveB = bStunned ? { action: 'idle' } : bAction;
        
        // === 第2步：同时执行移动（不冲突的情况下） ===
        const moveResults = resolveMovements(playerA, effectiveA, playerB, effectiveB);
        
        // === 第3步：同时执行攻击/技能 ===
        const combatResults = resolveCombat(playerA, effectiveA, playerB, effectiveB);
        
        // === 第4步：应用持续效果（DOT/恢复） ===
        applyBuffsTick(playerA, playerB);
        
        // === 第5步：检查死亡 ===
        if (playerA.hp <= 0) playerA.isAlive = false;
        if (playerB.hp <= 0) playerB.isAlive = false;
        
        // === 第6步：记录日志并广播 ===
        battleLog.push({
            stepIndex: step,
            actions: [moveResults.a, moveResults.b, combatResults.a, combatResults.b].filter(Boolean),
            results: getHpResults(playerA, playerB)
        });
        
        broadcastBattleAction(roomId, battleLog[battleLog.length - 1]);
        
        if (!playerA.isAlive || !playerB.isAlive) break;
    }
    
    // === 回合结束：减冷却、清队列 ===
    endRound(playerA, playerB);
    return battleLog;
}
```

## 四、具体判定逻辑

### 4.1 移动判定

```javascript
function resolveMovements(pA, actionA, pB, actionB) {
    const aMoves = actionA.action === 'move';
    const bMoves = actionB.action === 'move';
    
    if (aMoves) {
        const newPos = getNewPosition(pA.position, actionA.direction);
        if (isInBounds(newPos) && !(bMoves && samePosition(newPos, getNewPosition(pB.position, actionB.direction)))) {
            pA.position = newPos;
        }
    }
    if (bMoves) {
        const newPos = getNewPosition(pB.position, actionB.direction);
        if (isInBounds(newPos) && !(aMoves && samePosition(newPos, getNewPosition(pA.position, actionA.direction)))) {
            pB.position = newPos;
        }
    }
    // 两人移动到同一格时，都失败
}
```

### 4.2 普通攻击判定

```javascript
function resolveAttack(attacker, defender) {
    // 距离检查：攻击范围 = 1 格（相邻）
    if (getDistance(attacker.position, defender.position) > 1) {
        return { hit: false, reason: 'out_of_range' };
    }
    
    // 防御减免
    const defenderDefending = defender.isDefending;
    const defenseMultiplier = defenderDefending ? 0.4 : 1.0;
    
    // 伤害计算
    let damage = attacker.baseAttack;
    
    // Buff 修正
    damage *= getAttackMultiplier(attacker);      // 攻击方攻击buff
    damage *= getDefenseMultiplier(defender);      // 防御方防御buff
    damage *= defenseMultiplier;                    // 防御行动减免
    damage -= defender.baseDefense * 0.3;           // 基础防御减伤
    
    damage = Math.max(1, Math.round(damage));       // 最少1点伤害
    
    defender.hp -= damage;
    attacker.stats.damageDealt += damage;
    defender.stats.damageTaken += damage;
    
    return { hit: true, damage, to: defender.id };
}
```

### 4.3 技能释放判定

```javascript
function resolveSkill(caster, target, skillData) {
    // 冷却检查
    if (caster.skillCooldowns[skillData.id] > 0) {
        return { success: false, reason: 'on_cooldown' };
    }
    
    // 距离检查（如果技能有 range）
    if (skillData.range && getDistance(caster.position, target.position) > skillData.range) {
        return { success: false, reason: 'out_of_range' };
    }
    
    const result = { success: true, skillId: skillData.id, caster: caster.id };
    
    switch (skillData.type) {
        case 'attack':
            const dmg = calculateSkillDamage(caster, target, skillData);
            target.hp -= dmg;
            caster.stats.damageDealt += dmg;
            result.damage = dmg;
            result.target = target.id;
            
            // 击退
            if (skillData.knockback) {
                applyKnockback(caster, target, skillData.knockback);
            }
            // 吸血
            if (skillData.lifeSteal) {
                const heal = Math.round(dmg * skillData.lifeSteal);
                caster.hp = Math.min(caster.maxHp, caster.hp + heal);
            }
            break;
            
        case 'defense':
            target.isDefending = true;
            break;
            
        case 'heal':
            const healAmount = skillData.healAmount || 0;
            caster.hp = Math.min(caster.maxHp, caster.hp + healAmount);
            caster.stats.healingDone += healAmount;
            result.heal = healAmount;
            break;
            
        case 'movement':
            // 瞬间移动 skillData.distance 格
            // 具体方向由玩家当前朝向决定（或随机选择可行方向）
            break;
            
        case 'buff':
            applyBuffs(caster, skillData.effects);
            result.effects = skillData.effects;
            break;
            
        case 'debuff':
            applyBuffs(target, skillData.effects);
            result.effects = skillData.effects;
            result.target = target.id;
            break;
    }
    
    // 自伤
    if (skillData.selfDamage > 0) {
        caster.hp -= skillData.selfDamage;
    }
    
    // 设置冷却
    caster.skillCooldowns[skillData.id] = skillData.cooldown;
    caster.stats.skillsUsed++;
    
    return result;
}
```

### 4.4 Buff 系统

```javascript
function applyBuffs(player, effects) {
    for (const effect of effects) {
        // 检查是否已有同类型buff，有则刷新
        const existing = player.buffs.find(b => b.type === effect.type);
        if (existing) {
            existing.remainingDuration = effect.duration;
            existing.value = effect.value;
        } else {
            player.buffs.push({
                type: effect.type,
                value: effect.value || 0,
                remainingDuration: effect.duration
            });
        }
    }
}

function applyBuffsTick(playerA, playerB) {
    for (const player of [playerA, playerB]) {
        for (const buff of player.buffs) {
            switch (buff.type) {
                case 'poison':
                case 'burn':
                case 'bleed':
                    const dot = buff.value || buff.damagePerTurn || 5;
                    player.hp -= dot;
                    break;
                // 治疗类DOT理论上也可以有
            }
            buff.remainingDuration--;
        }
        // 移除过期buff
        player.buffs = player.buffs.filter(b => b.remainingDuration > 0);
    }
}

function getAttackMultiplier(player) {
    let mult = 1.0;
    for (const b of player.buffs) {
        if (b.type === 'attack_up') mult *= (1 + b.value / 100);
        if (b.type === 'attack_down') mult *= (1 - b.value / 100);
    }
    return mult;
}

// getDefenseMultiplier, getSpeedMultiplier 同理
```

## 五、伤害计算公式

### 普通攻击
```
damage = baseAttack × attackBuff × (1 - defenseReduction) × defenseBuff × defendMultiplier - baseDefense×0.3
最低伤害: 1
```

### 技能攻击
```
damage = skillDamage × attackBuff × (1 - defenseReduction) × defenseBuff × defendMultiplier
若 ignoreDefense: defenseReduction = 0, defendMultiplier = 1
最低伤害: 1
```

其中：
- `attackBuff`：攻击方的攻击buff累积（默认 1.0）
- `defenseReduction`：防御方的伤害减免buff（默认 0）
- `defenseBuff`：防御方的防御buff影响（默认 1.0）
- `defendMultiplier`：防御行动减免 0.4，否则 1.0

## 七、回合结束清理

```javascript
function endRound(playerA, playerB) {
    for (const player of [playerA, playerB]) {
        // 清空队列
        player.actionQueue = [];
        player.queueConfirmed = false;
        
        // 防御状态重置
        player.isDefending = false;
        
        // 冷却-1
        for (const skillId in player.skillCooldowns) {
            if (player.skillCooldowns[skillId] > 0) {
                player.skillCooldowns[skillId]--;
            }
        }
    }
}
```

## 八、胜负判定

```javascript
function checkWinner(playerA, playerB) {
    if (!playerA.isAlive && !playerB.isAlive) return 'draw';
    if (!playerA.isAlive) return playerB.id;
    if (!playerB.isAlive) return playerA.id;
    return null; // 继续
}
```
